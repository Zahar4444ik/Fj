header       ::= "type:" TYPE NEWLINE
sections     ::= (alphabet_section | states_section | start_section | accepting_section | transitions_section)+

alphabet_section ::= "alphabet:" "{" sym_list "}" NEWLINE
sym_list     ::= SYMBOL {"," SYMBOL}

states_section ::= "states:" "{" state_list "}" NEWLINE
state_list   ::= STATE {"," STATE}

start_section ::= "start:" STATE NEWLINE
accepting_section ::= "accepting:" "{" state_list "}" NEWLINE

transitions_section ::= "transitions:" NEWLINE transition_line {transition_line}
transition_line ::= [INDENT] TRANS_FROM INDICATOR TRANS_TO NEWLINE
TRANS_FROM   ::= STATE
INDICATOR    ::= "-" TRANS_SYMBOL "->"
TRANS_SYMBOL ::= "ε" | "eps" | SYMBOL
TRANS_TO     ::= STATE
TYPE         ::= "NFA" | "DFA"

# lexical tokens:
IDENT        ::= 'q' | 's'
STATE        ::= IDENT DIGITS
SYMBOL       ::= any single non-space printable char or multi-char token like "ab" depending on alphabet rules
COMMENT      ::= "#" .* NEWLINE
NEWLINE      ::= "\n" | "\r\n"
INDENT       ::= whitespace (spaces or tabs)
DIGITS       ::= {0-9}
